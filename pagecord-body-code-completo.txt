  if (!host || host.querySelector(".sb")) return;
  document.body.classList.add("sb-page");

  // Long-press menu: Download / Share / Copy link
  var menu = document.createElement("dialog");
  menu.className = "sb-menu";
  menu.innerHTML = '<p></p><button data-a="save">Download</button><button data-a="share">Share</button><button data-a="copy">Copy link</button><button data-a="close">Close</button>';
  var chosen;

  function getFile(s) {
    return fetch(s[1]).then(function (r) { return r.blob(); }).then(function (b) {
      return new File([b], s[0] + ".mp3", { type: "audio/mpeg" });
    });
  }
  function shareLink(s) {
    if (navigator.share) navigator.share({ title: s[0], url: s[1] });
    else navigator.clipboard.writeText(s[1]);
  }

  menu.onclick = function (e) {
    var a = e.target.getAttribute("data-a");
    if (!a && e.target !== menu) return;
    var s = chosen;
    if (a === "save") {
      getFile(s).then(function (f) {
        var l = document.createElement("a");
        l.href = URL.createObjectURL(f);
        l.download = f.name;
        l.click();
      }).catch(function () { window.open(s[1], "_blank"); });
    } else if (a === "share") {
      getFile(s).then(function (f) {
        if (navigator.canShare && navigator.canShare({ files: [f] })) navigator.share({ files: [f] });
        else shareLink(s);
      }).catch(function () { shareLink(s); });
    } else if (a === "copy") {
      navigator.clipboard.writeText(s[1]);
    }
    menu.close();
  };

  var box = document.createElement("div");
  box.className = "sb";
  var current, timer, held;

  sounds.forEach(function (s) {

    b.onclick = function () {
      if (held) { held = false; return; }   // this click was the end of a long press
      if (current) current.pause();
      audio.currentTime = 0;
      audio.play();
      current = audio;
    };

    b.onpointerdown = function () {
      held = false;
      timer = setTimeout(function () {
        held = true;
        chosen = s;
        menu.querySelector("p").textContent = s[0];
        menu.showModal();
      }, 500);
    };
    b.onpointerup = b.onpointerleave = b.onpointercancel = function () { clearTimeout(timer); };
    b.oncontextmenu = function (e) { e.preventDefault(); };

    box.appendChild(b);
  });

  host.appendChild(box);
  host.appendChild(menu);
})();
</script>

<script>/* WDC Radio 1.0 - player casuale per /radio */
(function () {
  'use strict';
  const SLUG = '/radio';
  // Coda da radio-tracks.js
  const TRACKS = window.OLIO_TRACKS || [];
  let cleanup = null;
  const shuffled = items => {
    const result = items.slice();
    for (let i=result.length-1;i>0;i--) {
      const j=Math.floor(Math.random()*(i+1));
      [result[i],result[j]]=[result[j],result[i]];
    }
    return result;
  };
  function init() {
    if (cleanup) {cleanup();cleanup=null;}
    if (location.pathname.replace(/\/$/,'') !== SLUG) return;
    let root=document.querySelector('[data-olio-radio]');
    if (!root) {
      const host=document.querySelector('article .post-body');
      if (!host) return;
      host.insertAdjacentHTML('beforeend', `
<section class="olio-radio" data-olio-radio aria-label="WDC Radio">
  <div class="olio-studio"><picture><source media="(max-width:740px)" srcset="https://oliodnb.github.io/soundboard/scrivania-mobile-stretta.jpg"><img class="olio-desk" src="https://oliodnb.github.io/soundboard/scrivania2.jpg" width="1170" height="600" alt="The WDC Radio desk, with monitor and instruments"></picture><div class="olio-video"><div data-player></div></div></div><p class="olio-note">Might show ads - <a href="https://fmhy.net/privacy">Adblocker</a> recommended</p>
  <h3 data-title>The selection is on its way</h3>
  <p data-status role="status" aria-live="polite">Loading...</p>
  <div class="olio-controls"><button type="button" data-back aria-label="Previous track">|&lt;</button><button type="button" data-play aria-label="Play">&gt;</button><button type="button" data-next aria-label="Next track">&gt;|</button><button type="button" data-mix aria-label="Shuffle">♻︎</button></div>
  <details class="olio-queue"><summary data-count>Queue</summary><ol data-list aria-label="Tracks in the queue"></ol></details>
  <footer class="olio-share" aria-label="Share WDC Radio">
    <a href="https://olio.pagecord.com/radio" aria-label="Open WDC Radio"><img class="olio-qr" src="https://oliodnb.github.io/soundboard/qr-radio.png" width="296" height="296" alt="QR code for https://olio.pagecord.com/radio"></a>
    <p><a href="https://olio.pagecord.com/radio">olio.pagecord.com/radio</a></p>
  </footer>
</section>
`);
      root=host.querySelector('[data-olio-radio]');
    }
    let player=null, ready=false, loading=false, dead=false, current=0, ended=false;
    let history=[];
    const tracks=TRACKS.filter(t=>/^[A-Za-z0-9_-]{11}$/.test(t.id));
    let queue=shuffled(tracks);
    const title=root.querySelector('[data-title]');
    const status=root.querySelector('[data-status]');
    const list=root.querySelector('[data-list]');
    const back=root.querySelector('[data-back]');
    const play=root.querySelector('[data-play]');
    const next=root.querySelector('[data-next]');
    const mix=root.querySelector('[data-mix]');
    const listeners=[];
    const on=(el,ev,fn)=>{el.addEventListener(ev,fn);listeners.push(()=>el.removeEventListener(ev,fn));};
    const message=text=>{status.textContent=text;}
    const byId=t=>player.loadVideoById(t.start!=null||t.end!=null?{videoId:t.id,startSeconds:t.start||0,endSeconds:t.end}:t.id);;
    function render(){
      list.replaceChildren();
      root.querySelector("[data-count]").textContent="Queue ("+queue.length+")";
      queue.forEach((track,i)=>{
        const li=document.createElement('li');
        const a=document.createElement('a');
        a.href='https://www.youtube.com/watch?v='+track.id;
        a.target='_blank';a.rel='noopener';a.textContent=track.title;
        li.append(a);if(i===current) li.setAttribute('aria-current','true');
        list.append(li);
      });
      title.textContent=queue.length ? queue[current].title : 'Track list unavailable';
      play.disabled=!queue.length; next.disabled=queue.length<2; mix.disabled=queue.length<2; back.disabled=!history.length;
    }
    function setPlay(icon,label){play.textContent=icon;play.setAttribute('aria-label',label);}
    function load(index){
      current=index;ended=false;render();
      if(ready){byId(queue[current]);message('Loading the video...');}
    }
    function advance(){
      if(current+1<queue.length){history.push(current);load(current+1);}
      else {ended=true;message('End of the selection. Press Play to listen again.');setPlay('>','Restart');}
    }
    async function api(){
      if(window.YT && window.YT.Player) return;
      if(!window.olioYouTubeReady) window.olioYouTubeReady=new Promise((resolve,reject)=>{
        const previous=window.onYouTubeIframeAPIReady;
        window.onYouTubeIframeAPIReady=()=>{if(typeof previous==='function')previous();resolve();};
        const s=document.createElement('script');s.src='https://www.youtube.com/iframe_api';
        s.onerror=()=>{window.olioYouTubeReady=null;reject(new Error('YouTube non raggiungibile'));};
        document.head.append(s);
      });
      await window.olioYouTubeReady;
    }
    on(play,'click',async()=>{
      if(!queue.length || loading)return;
      if(ready){
        if(player.getPlayerState()===1)player.pauseVideo();
        else if(ended){history=[];load(0);}
        else player.playVideo();
        return;
      }
      loading=true;message('Connecting to YouTube...');
      try {
        await api(); if(dead)return;
        player=new window.YT.Player(root.querySelector('[data-player]'),{
          width:'100%',height:'250',videoId:queue[current].id,
          host:'https://www.youtube-nocookie.com',
          playerVars:{playsinline:1,controls:1,origin:location.origin,...(queue[current].start!=null?{start:queue[current].start}:{}),...(queue[current].end!=null?{end:queue[current].end}:{})},
          events:{
            onReady:()=>{if(dead)return;ready=true;loading=false;const t=queue[current];if(player.getVideoData().video_id!==t.id)byId(t);else player.playVideo();message('Playing on YouTube');},
            onStateChange:e=>{
              if(dead)return;
              if(e.data===1){setPlay('||','Pause');message('Playing on YouTube');}
              if(e.data===2){setPlay('>','Play');message('Paused');}
              if(e.data===0){setPlay('>','Play');advance();}
            },
            onAutoplayBlocked:()=>{setPlay('>','Play');message('Tap Play in the YouTube frame to start.');},
            onError:e=>{loading=false;setPlay('>','Play');message('Video not available in the player (code '+e.data+'). Open it on YouTube or press Next.');}
          }
        });
      } catch(e){loading=false;setPlay('>','Play');message('Could not load YouTube. Try again or use the video link.');}
    });
    on(next,'click',()=>{if(queue.length){history.push(current);load((current+1)%queue.length);}});
    on(back,'click',()=>{if(history.length)load(history.pop());});
    on(mix,'click',()=>{
      // Shuffle senza ripetizioni; il brano in corso resta.
      if(!queue.length)return;
      history=[];
      if(ready) {queue=[queue[current],...shuffled(queue.filter((_,i)=>i!==current))];current=0;}
      else {queue=shuffled(queue);current=0;}
      render();message('Order shuffled. No repeats until the queue ends.');
    });
    render();
    message(queue.length?'':'Could not load the track list. Check your connection and reload.');
    cleanup=()=>{dead=true;listeners.forEach(fn=>fn());if(player&&player.destroy){player.destroy();const mount=document.createElement('div');mount.setAttribute('data-player','');root.querySelector('.olio-video').replaceChildren(mount);}};
  }
  document.addEventListener('turbo:load',init);
  document.addEventListener('turbo:before-cache',()=>{if(cleanup){cleanup();cleanup=null;}});
  if(document.readyState==='loading')document.addEventListener('DOMContentLoaded',init,{once:true});else init();
})();</script>
